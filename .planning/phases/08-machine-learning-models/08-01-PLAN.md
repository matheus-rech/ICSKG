```sql
-- dimension_scores table (Phase 5 output)
CREATE TABLE IF NOT EXISTS dimension_scores (
    cod_ibge       TEXT    NOT NULL,
    year           INTEGER NOT NULL,
    D1             REAL,   -- Governance (IFGF)
    D2             REAL,   -- Economy (GDP per capita)
    D3             REAL,   -- Human Capital (IDHM)
    D4             REAL,   -- Social Cohesion (SUS dependence, INVERTED)
    D5             REAL,   -- Environment (sanitation)
    D6             REAL,   -- Mobility (vehicles)
    D7             REAL,   -- Urban Planning (hospital beds)
    D9             REAL,   -- Technology (health expenditure)
    cuds           REAL,
    n_dimensions   INTEGER,
    PRIMARY KEY (cod_ibge, year)
);

-- lcogs_indicators table (Phase 6 output)
CREATE TABLE IF NOT EXISTS lcogs_indicators (
    cod_ibge                 TEXT    NOT NULL,
    year                     INTEGER NOT NULL,
    lcogs1_distance_km       REAL,
    sao_per_100k             REAL,
    surgical_volume_per_100k REAL,
    pomr                     REAL,
    financial_risk_ratio     REAL,
    catastrophic_expenditure REAL,
    PRIMARY KEY (cod_ibge, year)
);

-- municipality_lookup table (has region column for CV stratification)
CREATE TABLE IF NOT EXISTS municipality_lookup (
    cod_ibge   TEXT PRIMARY KEY,
    name       TEXT,
    state_uf   TEXT,
    region     TEXT   -- Norte, Nordeste, Centro-Oeste, Sudeste, Sul
);
```

New build-database.yml structure (to write):
```yaml
name: Build Database
on:
  workflow_dispatch:
    inputs:
      hf_revision:
        description: "HuggingFace dataset revision to build from (e.g., v0.1.0 or main)"
        required: false
        default: "v0.1.0"
        type: string
      start_year:
        description: "Start year (inclusive)"
        required: false
        default: "2015"
        type: string
      end_year:
        description: "End year (inclusive)"
        required: false
        default: "2023"
        type: string
      run_tests:
        description: "Run pytest before building"
        required: false
        default: true
        type: boolean

permissions:
  contents: read

concurrency:
  group: build-database-${{ inputs.hf_revision }}
  cancel-in-progress: false

jobs:
  build:
    name: Build ICSKG-BR Database from HF @ ${{ inputs.hf_revision }}
    runs-on: ubuntu-latest
    timeout-minutes: 30
    steps:
      - name: Checkout repository
        uses: actions/checkout@v5  # was v4

      - name: Set up Python 3.12
        uses: actions/setup-python@v5
        with:
          python-version: "3.12"
          cache: "pip"

      - name: Install project
        run: |
          python -m pip install --upgrade pip
          python -m pip install -e .
          python -m pip install dependency-groups
          pip-install-dependency-groups dev

      - name: Run tests
        if: ${{ inputs.run_tests }}
        run: python -m pytest tests/ -v --tb=short

      - name: Restore HuggingFace dataset cache
        id: hf-cache
        uses: actions/cache@v4
        with:
          path: ~/.cache/huggingface
          key: hf-icskg-br-processed-${{ inputs.hf_revision }}-${{ runner.os }}
          restore-keys: |
            hf-icskg-br-processed-${{ inputs.hf_revision }}-
            hf-icskg-br-processed-

      - name: Validate input years
        env:
          START_YEAR: ${{ inputs.start_year }}
          END_YEAR: ${{ inputs.end_year }}
        run: |
          [[ "$START_YEAR" =~ ^[0-9]{4}$ ]] || { echo "Invalid start_year: $START_YEAR" >&2; exit 1; }
          [[ "$END_YEAR" =~ ^[0-9]{4}$ ]] || { echo "Invalid end_year: $END_YEAR" >&2; exit 1; }

      - name: Build database from HuggingFace dataset
        env:
          HF_TOKEN: ${{ secrets.HF_TOKEN }}
          START_YEAR: ${{ inputs.start_year }}
          END_YEAR: ${{ inputs.end_year }}
          HF_REVISION: ${{ inputs.hf_revision }}
        run: |
          python -m database.build_database_v3 \
            --from-hf "$HF_REVISION" \
            --hf-repo-id "matheus-rech/icskg-br-processed" \
            --processed-dir data_sources/processed \
            --db-dir database \
            --years "$START_YEAR" "$END_YEAR"

      - name: Verify build outputs
        run: |
          test -f database/icskg_br.sqlite || { echo "Missing icskg_br.sqlite" >&2; exit 1; }
          python -c "
          import sqlite3
          conn = sqlite3.connect('database/icskg_br.sqlite')
          n = conn.execute('SELECT COUNT(*) FROM municipal_health').fetchone()[0]
          print(f'municipal_health rows: {n}')
          assert n > 0, 'municipal_health is empty — phantom database!'
          assert n == 50130, f'Expected 50130 rows, got {n}'
          "

      - name: Upload database artifact
        uses: actions/upload-artifact@v5  # was v4
        with:
          name: icskg-br-database-${{ inputs.hf_revision }}
          path: database/icskg_br.sqlite
          retention-days: 30
          if-no-files-found: error
```

```yaml
name: Build Database (smoke)

on:
  push:
    branches:
      - meta
      - feature/**
    paths:
      - "database/**"
      - "scripts/**"
      - "tests/**"
      - "pyproject.toml"
      - "uv.lock"
      - ".github/workflows/build-database-smoke.yml"
  pull_request:
    branches:
      - meta
    paths:
      - "database/**"
      - "scripts/**"
      - "tests/**"
      - "pyproject.toml"
      - "uv.lock"

permissions:
  contents: read

concurrency:
  group: build-database-smoke-${{ github.ref }}
  cancel-in-progress: true

jobs:
  smoke:
    name: Smoke build (committed fixture, no HF)
    runs-on: ubuntu-latest
    timeout-minutes: 10
    steps:
      - name: Checkout repository
        uses: actions/checkout@v5

      - name: Set up Python 3.12
        uses: actions/setup-python@v5
        with:
          python-version: "3.12"
          cache: "pip"

      - name: Install project
        run: |
          python -m pip install --upgrade pip
          python -m pip install -e .
          python -m pip install dependency-groups
          pip-install-dependency-groups dev

      - name: Run unit tests
        run: python -m pytest tests/ -v --tb=short -x

      - name: Smoke build against committed fixture
        run: |
          python -m database.build_database_v3 \
            --processed-dir tests/fixtures/processed_smoke \
            --db-dir /tmp/icskg_smoke \
            --years 2022 2023

      - name: Verify smoke outputs
        run: |
          test -f /tmp/icskg_smoke/icskg_br.sqlite || { echo "Missing smoke sqlite" >&2; exit 1; }
          python -c "
          import sqlite3
          conn = sqlite3.connect('/tmp/icskg_smoke/icskg_br.sqlite')
          n = conn.execute('SELECT COUNT(*) FROM municipal_health').fetchone()[0]
          assert n == 10, f'Expected 10 smoke rows, got {n}'
          print(f'Smoke build OK: {n} rows')
          "

      - name: Verify fail-loud guard works on empty input
        run: |
          mkdir -p /tmp/empty_processed
          set +e
          python -m database.build_database_v3 \
            --processed-dir /tmp/empty_processed \
            --db-dir /tmp/icskg_empty \
            --years 2022 2023 2>&1 | tee /tmp/guard_output.log
          rc=$?
          set -e
          if [ "$rc" -eq 0 ]; then
            echo "ERROR: build_database_v3 succeeded on empty input — fail-loud guard is broken" >&2
            exit 1
          fi
          grep -q "Refusing to build a phantom database" /tmp/guard_output.log || {
            echo "ERROR: expected guard message not found in output" >&2
            exit 1
          }
          echo "Fail-loud guard verified."
```

<gotchas>
- `actions/upload-artifact@v5` has a breaking change vs v4: `name` collisions across jobs error out instead of merging. We use a unique name per revision so this is fine, but worth noting in case the workflow ever gets parallelized.
- `actions/cache@v4` is the latest stable. Cache key MUST include `runner.os` because the parquet files are platform-independent but pip caches are not.
- HF cache directory is `~/.cache/huggingface/hub/` (with `hub/` subdir). Caching the parent `~/.cache/huggingface` is broader but safer — covers token cache too which is fine because we get a fresh token from secrets every run.
- The `hf_revision` input default is `v0.1.0` to start. Once Phase 11.1 publishes a newer revision, the user manually changes this default and re-dispatches.
- The smoke workflow MUST NOT fetch from HF — that's the point. Verify by running with no `HF_TOKEN` env var.
- `actions/cache@v4` cache key uses `${{ inputs.hf_revision }}` which means each revision gets its own cache slot. If the user re-runs with the same revision, the cache hits and skips the download entirely (free, fast).
- The `Verify build outputs` step in the full workflow asserts exactly 50130 rows (5570 × 9). If a future revision has different scope, that assertion needs to update — but until then, "exactly 50130" is the right invariant.
- `concurrency.cancel-in-progress: false` on the full workflow prevents accidental cancellation of an in-progress real build. The smoke workflow uses `cancel-in-progress: true` because superseding pushes should obviously cancel old smoke runs.
- The smoke workflow paths filter excludes `analysis/**` deliberately — analysis changes don't affect the build pipeline. Adjust if that proves wrong.
- Branch protection on `meta` should require `Build Database (smoke)` to pass before merge. That setup is manual via the GitHub UI — note in the SUMMARY for the user.
</gotchas>
